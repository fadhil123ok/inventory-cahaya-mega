  <iframe
    id="gasAppFrame"
    title="Cahaya Mega — The Paint Specialist"
    class="block w-full h-[100dvh] border-0 bg-canvas"
    allow="clipboard-read; clipboard-write; camera; microphone"
    allowfullscreen

  <script>
    /**
     * ========================================================================
     * KONFIGURASI UTAMA FILE PEMBUNGKUS (GITHUB & VERCEL)
     * ========================================================================
     * Ganti teks di bawah ini dengan URL Aplikasi Web Google Apps Script Anda
     * yang berakhiran "/exec" (dari proyek Google Sheets yang memasang penghubung.gs).
     * Contoh:
     * var GAS_WEB_APP_URL = 'https://script.google.com/macros/s/AKfycbxxxxxxxx/exec';
     */
    var GAS_WEB_APP_URL = 'PASTE_URL_DEPLOY_WEB_APP_APPS_SCRIPT_ANDA_DI_SINI';

    var LOCAL_URL_KEY = 'CM_VERCEL_WRAPPER_GAS_URL';

    function isValidGasUrl(url) {
      var u = String(url || '').trim();
      return /^https:\/\/script\.google\.com\/macros\/s\/[a-zA-Z0-9_\-]+\/(exec|dev)(\?.*)?$/.test(u);
    }

    function hideSplash() {
      var splash = document.getElementById('splashLoader');
      if (!splash) return;
      splash.classList.add('opacity-0', 'pointer-events-none');
      setTimeout(function() {
        splash.classList.add('hidden');
      }, 520);
    }

    function mountGasIframe(targetUrl) {
      var setupPanel = document.getElementById('setupPanel');
      var splash = document.getElementById('splashLoader');
      var frame = document.getElementById('gasAppFrame');

      setupPanel.classList.add('hidden');
      splash.classList.remove('hidden', 'opacity-0', 'pointer-events-none');

      // Teruskan query string bila ada (opsional)
      var finalUrl = targetUrl;
      if (window.location.search && targetUrl.indexOf('?') === -1) {
        finalUrl = targetUrl + window.location.search;
      }

      var loaded = false;
      frame.onload = function() {
        loaded = true;
        hideSplash();
      };

      // Pengaman jika onload iframe tertunda di beberapa browser mobile
      setTimeout(function() {
        if (!loaded) hideSplash();
      }, 6500);

      frame.src = finalUrl;
    }

    function showSetupPanel() {
      hideSplash();
      document.getElementById('setupPanel').classList.remove('hidden');
    }

    function applyQuickUrl() {
      var input = document.getElementById('quickGasUrl');
      var errBox = document.getElementById('quickUrlError');
      var val = String(input.value || '').trim();

      errBox.classList.add('hidden');
      if (!isValidGasUrl(val)) {
        errBox.textContent = 'Format URL tidak valid. Pastikan URL diawali https://script.google.com/macros/s/... dan diakhiri /exec.';
        errBox.classList.remove('hidden');
        return;
      }

      try {
        localStorage.setItem(LOCAL_URL_KEY, val);
      } catch (e) {}

      mountGasIframe(val);
    }

    function resetSavedUrl() {
      try {
        localStorage.removeItem(LOCAL_URL_KEY);
      } catch (e) {}
      document.getElementById('quickGasUrl').value = '';
      var errBox = document.getElementById('quickUrlError');
      errBox.textContent = 'URL tersimpan di browser ini telah dihapus.';
      errBox.classList.remove('hidden');
    }

    window.addEventListener('DOMContentLoaded', function() {
      try {
        if (isValidGasUrl(GAS_WEB_APP_URL)) {
          mountGasIframe(GAS_WEB_APP_URL);
          return;
        }

        var savedUrl = '';
        try {
          savedUrl = localStorage.getItem(LOCAL_URL_KEY) || '';
        } catch (e) {}

        if (isValidGasUrl(savedUrl)) {
          mountGasIframe(savedUrl);
        } else {
          showSetupPanel();
        }
      } catch (err) {
        showSetupPanel();
      }
    });
  </script>
